cmake_minimum_required(VERSION 3.16)
project(gesture_tflite LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Compile against the GCC 9 ARM standard headers matching the board's
# libstdc++.so.6.0.27 ABI, rather than the compiler's GCC 13 headers.
set(CPP9 "${CMAKE_CURRENT_SOURCE_DIR}/cpp9_headers/usr/arm-linux-gnueabihf/include/c++/9")
if(NOT EXISTS "${CPP9}/arm-linux-gnueabihf/bits/c++config.h")
  message(FATAL_ERROR "Run prepare_cpp9.sh before configuring")
endif()
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -nostdinc++ -isystem ${CPP9} -isystem ${CPP9}/arm-linux-gnueabihf -isystem ${CPP9}/backward")
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-include>" "$<$<COMPILE_LANGUAGE:CXX>:cstdint>")
option(GESTURE_USE_XNNPACK "Enable the explicit Cortex-A9 XNNPACK delegate" OFF)
set(TFLITE_ENABLE_XNNPACK ${GESTURE_USE_XNNPACK} CACHE BOOL "" FORCE)
set(XNNPACK_ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
set(XNNPACK_ENABLE_JIT OFF CACHE BOOL "" FORCE)
set(XNNPACK_ENABLE_ARM_FP16 OFF CACHE BOOL "" FORCE)
set(XNNPACK_ENABLE_ARM_BF16 OFF CACHE BOOL "" FORCE)
set(XNNPACK_ENABLE_ARM_DOTPROD OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_RUY OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_NNAPI OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_EXTERNAL_DELEGATE OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_RESOURCE ON CACHE BOOL "" FORCE)
set(TFLITE_KERNEL_TEST OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATC OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATHASH OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
add_subdirectory(tensorflow-2.11.0/tensorflow/lite tflite)
if(GESTURE_USE_XNNPACK)
  set(RUNTIME_TARGET gesture_tflite_xnnpack)
else()
  set(RUNTIME_TARGET gesture_tflite)
endif()
add_library(${RUNTIME_TARGET} SHARED gesture_tflite.cc)
target_link_libraries(${RUNTIME_TARGET} PRIVATE tensorflow-lite)
target_link_options(${RUNTIME_TARGET} PRIVATE "-Wl,-z,defs")
if(GESTURE_USE_XNNPACK)
  target_compile_definitions(${RUNTIME_TARGET} PRIVATE GESTURE_USE_XNNPACK=1)
endif()
option(GESTURE_BUILD_PL "Build the PL hardware and explicit C-simulation delegates" OFF)
if(GESTURE_BUILD_PL)
  add_subdirectory(pl_runtime)
endif()
