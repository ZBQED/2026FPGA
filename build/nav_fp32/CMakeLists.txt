cmake_minimum_required(VERSION 3.16)
project(gesture_nav_fp32 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm")
  message(FATAL_ERROR "Provide the target Cortex-A9 Linux toolchain; host builds are not deployment builds")
endif()
if(NOT EXISTS "${GESTURE_TENSORFLOW_SOURCE}/tensorflow/lite/CMakeLists.txt")
  message(FATAL_ERROR "GESTURE_TENSORFLOW_SOURCE must point to the TensorFlow v2.11.0 source tree")
endif()
add_compile_options(-mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -marm -ffp-contract=off)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -include cstdint -include limits")
# GCC 13 with the recorded vendor GCC 9 C++ ABI requires the matching headers.
if(GESTURE_CPP9_INCLUDE)
  if(NOT EXISTS "${GESTURE_CPP9_INCLUDE}/arm-linux-gnueabihf/bits/c++config.h")
    message(FATAL_ERROR "Invalid GESTURE_CPP9_INCLUDE")
  endif()
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -nostdinc++ -isystem ${GESTURE_CPP9_INCLUDE} -isystem ${GESTURE_CPP9_INCLUDE}/arm-linux-gnueabihf -isystem ${GESTURE_CPP9_INCLUDE}/backward")
endif()
foreach(flag IN ITEMS XNNPACK RUY NNAPI GPU EXTERNAL_DELEGATE)
  set(TFLITE_ENABLE_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(TFLITE_ENABLE_RESOURCE ON CACHE BOOL "" FORCE)
set(TFLITE_KERNEL_TEST OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATC OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATHASH OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
add_subdirectory("${GESTURE_TENSORFLOW_SOURCE}/tensorflow/lite" tflite)
get_filename_component(PACKAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SOURCE "${PACKAGE_ROOT}/src/native/nav_fp32")
add_library(gesture_tflite_pl SHARED
  "${SOURCE}/gesture_tflite.cc"
  "${SOURCE}/pl_runtime/pl_delegate.cc"
  "${SOURCE}/pl_runtime/gesture_pl_client.cpp"
  "${GESTURE_TENSORFLOW_SOURCE}/tensorflow/lite/delegates/utils/simple_delegate.cc")
target_include_directories(gesture_tflite_pl PRIVATE "${SOURCE}/pl_runtime" "${PACKAGE_ROOT}/src/hls/nav_fp32")
target_compile_definitions(gesture_tflite_pl PRIVATE GESTURE_USE_PL=1)
target_link_libraries(gesture_tflite_pl PRIVATE tensorflow-lite)
target_link_options(gesture_tflite_pl PRIVATE "-Wl,-z,defs")
